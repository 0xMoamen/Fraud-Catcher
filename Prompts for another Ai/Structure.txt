fraud-catcher/
│
├── backend/
│   ├── ml_engine/              # FEATURE 1: Traditional ML Model
│   │   ├── train.py            # Script to train XGBoost/Random Forest
│   │   ├── model.pkl           # Saved trained model
│   │   └── predictor.py        # Loads model and returns {score, confidence}
│   │
│   ├── llm_agents/             # FEATURE 2: Multi-Agent LLM System
│   │   ├── analyzer_agent.py   # Agent 1: Extracts and validates raw signals
│   │   ├── pattern_agent.py    # Agent 2: Matches against known fraud typologies
│   │   ├── judge_agent.py      # Agent 3: Synthesizes findings into final score/reasoning
│   │   └── crew.py             # Orchestrates the agents (using LangGraph or CrewAI)
│   │
│   └── orchestrator.py         # FEATURE 3: The "Router"
│       └── main.py             # FastAPI endpoint that accepts "mode": "ml" | "llm" | "ensemble"
│
├── chatbots/                   # FEATURE 4: Dual Chatbot System
│   ├── customer_bot.py         # Answers "Why was my transaction declined?" (Polite, safe)
│   └── admin_bot.py            # Answers "Investigate TX_1001" (Deep dive, pulls DB context)
│
├── frontend/                   # FEATURE 5: Modular Dashboard
│   ├── components/             # Reusable UI (ModelToggle.tsx, FraudTable.tsx, ChatWidget.tsx)
│   ├── pages/
│   │   ├── dashboard.tsx       # Main view with charts and transaction list
│   │   ├── investigate.tsx     # Admin view with deep-dive tools
│   │   └── customer_portal.tsx # Customer-facing status checker
│   └── package.json            # Next.js + Tailwind CSS (Copilot-friendly)
│
└── data/                       # Synthetic datasets, logs, and evaluation results
